cmake_minimum_required(VERSION 3.20)
project(upio LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Debug)
endif()

option(UPIO_SANITIZE "Build with ASan and UBSan" ON)
add_compile_options(-Wall -Wextra -Wpedantic)
if(UPIO_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

find_path(LIBURING_INCLUDE_DIR liburing.h REQUIRED)
find_library(LIBURING_LIBRARY uring REQUIRED)
add_library(liburing INTERFACE)
target_include_directories(liburing INTERFACE ${LIBURING_INCLUDE_DIR})
target_link_libraries(liburing INTERFACE ${LIBURING_LIBRARY})

enable_testing()

function(add_lab name)
  cmake_parse_arguments(LAB "URING" "" "" ${ARGN})
  add_executable(${name} lab/${name}.cc)
  if(LAB_URING)
    target_link_libraries(${name} PRIVATE liburing)
  endif()
  add_test(NAME ${name} COMMAND ${name})
endfunction()

add_lab(00_raw_ring)
add_lab(01_liburing_read URING)
add_lab(02_batch URING)
add_lab(03_link_drain URING)
add_lab(04_registered URING)
add_lab(05_modes URING)

add_library(upio src/reactor.cc src/files.cc src/pool.cc src/ring_server.cc)
target_include_directories(upio PUBLIC include)
find_package(Threads REQUIRED)
target_link_libraries(upio PUBLIC liburing Threads::Threads)

add_executable(m3_test tests/m3_test.cc)

add_executable(m4_test tests/m4_test.cc)
